1. Objetivo desta unidade

Nestas semanas vamos aprofundar o uso de decisões (if, switch) e repetições (while, for) sempre combinados com objetos.

A meta é que você consiga usar objetos para modelar situações reais (contas, pedidos, tarefas) e controlar o fluxo do programa com base nos atributos desses objetos.

Leia os exemplos, depois execute e modifique o código em um compilador online ou no seu VS Code. Os exemplos abaixo já podem ser executados diretamente na página via JDoodle.

2. Exemplo: Banco simples com if e while

Neste exemplo, criamos uma conta bancária e permitimos que o usuário faça depósitos e saques até decidir sair.

Código Java


import java.util.Scanner;

class ContaBancaria {
    String titular;
    double saldo;

    void depositar(double valor) {
        saldo = saldo + valor;
    }

    void sacar(double valor) {
        if (valor <= saldo) {
            saldo = saldo - valor;
        } else {
            System.out.println("Saldo insuficiente.");
        }
    }

    void exibirResumo() {
        System.out.println("Titular: " + titular + " | Saldo: R$ " + saldo);
    }
}

public class BancoSimples {

    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);

        ContaBancaria conta = new ContaBancaria();
        System.out.print("Nome do titular: ");
        conta.titular = entrada.nextLine();
        conta.saldo = 0.0;

        int opcao;
        do {
            System.out.println("\n=== MENU BANCO ===");
            System.out.println("1 - Depositar");
            System.out.println("2 - Sacar");
            System.out.println("3 - Exibir resumo");
            System.out.println("0 - Sair");
            System.out.print("Escolha: ");
            opcao = entrada.nextInt();

            if (opcao == 1) {
                System.out.print("Valor para depósito: ");
                double valor = entrada.nextDouble();
                conta.depositar(valor);
            } else if (opcao == 2) {
                System.out.print("Valor para saque: ");
                double valor = entrada.nextDouble();
                conta.sacar(valor);
            } else if (opcao == 3) {
                conta.exibirResumo();
            } else if (opcao != 0) {
                System.out.println("Opção inválida.");
            }

        } while (opcao != 0);

        System.out.println("Encerrando... Obrigado por usar o nosso banco!");
        entrada.close();
    }
}

3. Exemplo: Sistema de pedidos com switch e for

Agora vamos usar switch e um for para registrar vários pedidos de lanchonete, cada um representado por um objeto.

Código Java


import java.util.Scanner;

class Pedido {
    int codigoProduto; // 1 = Hambúrguer, 2 = Pizza, 3 = Suco
    int quantidade;
    double valorTotal;
}

public class Lanchonete {

    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);

        System.out.print("Quantos pedidos deseja registrar? ");
        int n = entrada.nextInt();

        Pedido[] pedidos = new Pedido[n];

        for (int i = 0; i < n; i++) {
            System.out.println("\n--- Pedido " + (i + 1) + " ---");
            pedidos[i] = new Pedido();

            System.out.println("1 - Hambúrguer (R$ 15.00)");
            System.out.println("2 - Pizza      (R$ 40.00)");
            System.out.println("3 - Suco       (R$  7.50)");
            System.out.print("Código do produto: ");
            pedidos[i].codigoProduto = entrada.nextInt();

            System.out.print("Quantidade: ");
            pedidos[i].quantidade = entrada.nextInt();

            double precoUnitario;

            switch (pedidos[i].codigoProduto) {
                case 1:
                    precoUnitario = 15.0;
                    break;
                case 2:
                    precoUnitario = 40.0;
                    break;
                case 3:
                    precoUnitario = 7.5;
                    break;
                default:
                    precoUnitario = 0.0;
                    System.out.println("Código inválido, valor será 0.");
            }

            pedidos[i].valorTotal = precoUnitario * pedidos[i].quantidade;
        }

        System.out.println("\n=== RESUMO DOS PEDIDOS ===");
        double somaTotal = 0.0;
        for (int i = 0; i < n; i++) {
            System.out.println("Pedido " + (i + 1) +
                    " | código: " + pedidos[i].codigoProduto +
                    " | qtd: " + pedidos[i].quantidade +
                    " | total: R$ " + pedidos[i].valorTotal);
            somaTotal += pedidos[i].valorTotal;
        }

        System.out.println("Valor total de todos os pedidos: R$ " + somaTotal);

        entrada.close();
    }
}
